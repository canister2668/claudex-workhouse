cmake_minimum_required(VERSION 3.25)
project(ClaudexWorkhouseLauncher LANGUAGES CXX RC)
add_executable(claudex-workhouse WIN32 src/main.cpp launcher.rc)
target_compile_features(claudex-workhouse PRIVATE cxx_std_20)
target_compile_definitions(claudex-workhouse PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX WINVER=0x0A00 _WIN32_WINNT=0x0A00)
target_link_libraries(claudex-workhouse PRIVATE bcrypt comctl32 winhttp shell32 ole32 user32 gdi32 advapi32 dwmapi)
if(MSVC)
  target_compile_options(claudex-workhouse PRIVATE /W4 /WX /utf-8 /guard:cf)
  target_link_options(claudex-workhouse PRIVATE /DYNAMICBASE /NXCOMPAT /HIGHENTROPYVA)
endif()
