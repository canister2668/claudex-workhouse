cmake_minimum_required(VERSION 3.25)
project(ClaudexWorkhouseLauncher LANGUAGES CXX RC)
add_executable(claudex-workhouse WIN32 src/main.cpp launcher.rc)
# Console helper that hosts Claude Code's TUI in a ConPTY for the server's
# usage, model-picker, cloud-session and login helpers (see src/conpty-bridge.cpp).
add_executable(claudex-conpty-bridge src/conpty-bridge.cpp)
foreach(target claudex-workhouse claudex-conpty-bridge)
  target_compile_features(${target} PRIVATE cxx_std_20)
  target_compile_definitions(${target} PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX WINVER=0x0A00 _WIN32_WINNT=0x0A00)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /WX /utf-8 /guard:cf)
    target_link_options(${target} PRIVATE /DYNAMICBASE /NXCOMPAT /HIGHENTROPYVA)
  endif()
endforeach()
target_link_libraries(claudex-workhouse PRIVATE bcrypt comctl32 winhttp shell32 ole32 user32 gdi32 advapi32 dwmapi)
